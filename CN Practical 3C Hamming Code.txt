#include <iostream>
using namespace std;
int main() {
    int h[8] = {0}, pos;
    cout << "Enter 4 data bits: ";
    cin >> h[3] >> h[5] >> h[6] >> h[7];

    h[1] = h[3] ^ h[5] ^ h[7];
    h[2] = h[3] ^ h[6] ^ h[7];
    h[4] = h[5] ^ h[6] ^ h[7];

    cout << "Hamming Code: ";
    for (int i = 1; i <= 7; i++)
        cout << h[i];
    cout << "\nEnter error position (0 for none): ";
    cin >> pos;
    if (pos < 0 || pos > 7) {
        cout << "Invalid position!";
        return 1;
    }
    
    if (pos != 0)
        h[pos] = h[pos] ^ 1;
    cout << "Received Code: ";
    for (int i = 1; i <= 7; i++)
        cout << h[i];

    int s1 = h[1] ^ h[3] ^ h[5] ^ h[7];
    int s2 = h[2] ^ h[3] ^ h[6] ^ h[7];
    int s4 = h[4] ^ h[5] ^ h[6] ^ h[7];

    int error = s1 + 2 * s2 + 4 * s4;
    if (error == 0)
        cout << "\nNo error detected.";
    else {
        cout << "\nError at position: " << error;
        h[error] = h[error] ^ 1;
        cout << "\nCorrected Code: ";
        for (int i = 1; i <= 7; i++)
            cout << h[i];
    }

    cout << "\nRecovered Data: ";
    cout << h[3] << h[5] << h[6] << h[7];
    return 0;
}
