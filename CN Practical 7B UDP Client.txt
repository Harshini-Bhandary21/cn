#include <iostream>
#include <cstring>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <unistd.h>
using namespace std;

int main()
{
    char buffer[1024] = {0};

    // 1. Create socket
    int clientSocket = socket(AF_INET, SOCK_DGRAM, 0);

    // 2. Configure server address
    sockaddr_in serverAddr{};
    socklen_t addrSize = sizeof(serverAddr);

    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(8081);

    inet_pton(AF_INET, "127.0.0.1",
              &serverAddr.sin_addr);

    // 3. Send message
    char msg[] = "Hello from UDP Client";

    sendto(clientSocket, msg, strlen(msg), 0,
           (sockaddr*)&serverAddr, addrSize);

    // 4. Receive message
    recvfrom(clientSocket, buffer, sizeof(buffer)-1, 0,
             (sockaddr*)&serverAddr, &addrSize);

    cout << "Server: " << buffer << endl;

    // 5. Close socket
    close(clientSocket);

    return 0;
}
